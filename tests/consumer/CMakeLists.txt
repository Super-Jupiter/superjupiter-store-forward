cmake_minimum_required(VERSION 3.16)
project(SuperJupiterStoreForwardConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_STORE_FORWARD_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_STORE_FORWARD_SOURCE_DIR}" store-forward)
  if(TARGET superjupiter-store-forward-tests OR TARGET superjupiter-store-forward-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterStoreForward 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-store-forward-consumer main.cpp)
target_link_libraries(superjupiter-store-forward-consumer PRIVATE SuperJupiter::StoreForward)
